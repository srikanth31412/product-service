# =============================
# STAGE 1: Build the application
# =============================
FROM eclipse-temurin:17-jdk-jammy AS build

# Set working directory
WORKDIR /workspace

# Copy Maven/Gradle wrapper and config
COPY mvnw .
COPY .mvn .mvn
COPY pom.xml .
# Download dependencies only (cache layer)
RUN ./mvnw dependency:go-offline

# Copy source code
COPY src src

# Build the application (skip tests for production image)
RUN ./mvnw clean package -DskipTests

# =============================
# STAGE 2: Create lightweight runtime image
# =============================
FROM eclipse-temurin:17-jre-jammy

# Set working directory
WORKDIR /app

# Copy the built jar from the previous stage
COPY --from=build /workspace/target/product-service-0.0.1-SNAPSHOT.jar app.jar

# Expose port
EXPOSE 8083

# Set JVM options (tune memory as needed)
ENV JAVA_OPTS="-Xms256m -Xmx512m"

# Run the Spring Boot application
ENTRYPOINT ["sh", "-c", "java $JAVA_OPTS -jar app.jar"]
